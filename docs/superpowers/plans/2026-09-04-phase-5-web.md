# Phase 5: Web App Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Next.js app where a team connects Meta, sets up a brand kit, browses creatives with tags and metrics, reads insights and fatigue alerts, chats with the Copilot, generates variants, reviews them, and publishes to Meta.

**Architecture:** Next.js App Router under `web/`, Clerk for auth, a thin typed API client that attaches either a Clerk JWT or dev headers, server components for reads and client components for interaction. shadcn/ui for primitives, Recharts for the metric chart. Vitest + Testing Library for components, one Playwright flow.

**Tech Stack:** Next.js 15, React 19, TypeScript, Tailwind, shadcn/ui, Clerk (`@clerk/nextjs`), Recharts, Vitest, Playwright.

**Spec:** `docs/superpowers/specs/2026-09-04-creative-strategist-design.md` (section 10)

## Global Constraints

- All API calls go through `web/src/lib/api.ts`; no bare `fetch` to the backend elsewhere.
- Auth modes: `NEXT_PUBLIC_AUTH_MODE=dev` sends `X-Dev-User`/`X-Dev-Org`/`X-Dev-Role` from `NEXT_PUBLIC_DEV_*` env; `clerk` sends `Authorization: Bearer <Clerk session token>` and requires an active Clerk organization.
- Backend URL from `NEXT_PUBLIC_API_URL` (default `http://localhost:8000`).
- Pages listed in spec section 10 all exist by the end of this phase.
- Node ≥ 22.

```ts
export async function authHeaders(): Promise<Record<string, string>> {
  if (process.env.NEXT_PUBLIC_AUTH_MODE !== 'clerk') {
    return {
      'X-Dev-User': process.env.NEXT_PUBLIC_DEV_USER ?? 'user_1',
      'X-Dev-Org': process.env.NEXT_PUBLIC_DEV_ORG ?? 'org_dev',
      'X-Dev-Role': process.env.NEXT_PUBLIC_DEV_ROLE ?? 'admin',
    };
  }
  let token: string | null | undefined;
  if (typeof window === 'undefined') {
    const { auth } = await import('@clerk/nextjs/server');
    token = await (await auth()).getToken();
  } else {
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    token = await (window as any).Clerk?.session?.getToken();
  }
  return token ? { Authorization: `Bearer ${token}` } : {};
}
```

```ts
import { authHeaders } from '@/lib/auth-headers';

export class ApiError extends Error {
  constructor(public status: number, message: string) { super(message); }
}

const base = () => (process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:8000').replace(/\/$/, '');

export async function apiFetch<T>(path: string, init: RequestInit = {}): Promise<T> {
  const headers: Record<string, string> = { ...(await authHeaders()), ...((init.headers as Record<string, string>) ?? {}) };
  if (typeof init.body === 'string' && !headers['Content-Type']) headers['Content-Type'] = 'application/json';
  const res = await fetch(`${base()}${path}`, { ...init, headers, cache: 'no-store' });
  if (!res.ok) {
    let msg = res.statusText;
    try { msg = (await res.json()).detail ?? msg; } catch { /* keep statusText */ }
    throw new ApiError(res.status, msg);
  }
  if (res.status === 204) return undefined as T;
  return (await res.json()) as T;
}
```

```tsx
'use client';
import { useState } from 'react';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import { Textarea } from '@/components/ui/textarea';
import { api, apiFetch } from '@/lib/api';
import type { BrandKit } from '@/lib/types';

const list = (s: string, sep: RegExp) => s.split(sep).map((x) => x.trim()).filter(Boolean);

export function BrandKitForm({ initial }: { initial: BrandKit | null }) {
  const [colors, setColors] = useState(initial?.colors.join(', ') ?? '');
  const [fonts, setFonts] = useState(initial?.fonts.join(', ') ?? '');
  const [voice, setVoice] = useState(initial?.voice_rules ?? '');
  const [prohibited, setProhibited] = useState(initial?.prohibited_claims.join('\n') ?? '');
  const [logoUrl, setLogoUrl] = useState(initial?.logo_url ?? null);
  const [msg, setMsg] = useState<string | null>(null);

  const save = async () => {
    const kit = await api.saveBrandKit({ colors: list(colors, /,/), fonts: list(fonts, /,/), voice_rules: voice, prohibited_claims: list(prohibited, /\n/) });
    setMsg(`Saved version ${kit.version}.`);
  };
  const upload = async (file: File) => {
    const fd = new FormData(); fd.append('file', file);
    const kit = await apiFetch<BrandKit>('/brand-kit/logo', { method: 'POST', body: fd });
    setLogoUrl(kit.logo_url); setMsg(`Logo saved (version ${kit.version}).`);
  };

  return (
    <section className="space-y-3">
      <h2 className="text-lg font-semibold">Brand kit</h2>
```

```tsx
'use client';
import { useEffect, useState } from 'react';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import { api } from '@/lib/api';
import type { Conversation, Message } from '@/lib/types';

export function CopilotChat({ conversations, initialId }: { conversations: Conversation[]; initialId: string | null }) {
  const [convs, setConvs] = useState(conversations);
  const [active, setActive] = useState<string | null>(initialId);
  const [messages, setMessages] = useState<Message[]>([]);
  const [input, setInput] = useState('');
  const [pending, setPending] = useState(false);

  useEffect(() => { if (active) api.messages(active).then(setMessages); else setMessages([]); }, [active]);

  const send = async () => {
    const content = input.trim();
    if (!content || pending) return;
    let id = active;
    if (!id) { const c = await api.newConversation(content.slice(0, 60)); id = c.id; setConvs([{ id, title: content.slice(0, 60), created_at: new Date().toISOString() }, ...convs]); setActive(id); }
    setMessages((m) => [...m, { role: 'user', content }]); setInput(''); setPending(true);
    try { const { reply } = await api.send(id, content); setMessages((m) => [...m, { role: 'assistant', content: reply }]); }
    finally { setPending(false); }
  };

  return (
    <div className="grid h-[70vh] grid-cols-4 gap-4">
      <aside className="space-y-2 overflow-y-auto border-r pr-3 text-sm">
        <Button size="sm" variant="outline" onClick={() => setActive(null)}>New conversation</Button>
        {convs.map((c) => <button key={c.id} onClick={() => setActive(c.id)} className={`block w-full truncate rounded px-2 py-1 text-left ${c.id === active ? 'bg-muted' : ''}`}>{c.title}</button>)}
      </aside>
      <section className="col-span-3 flex flex-col">
        <div className="flex-1 space-y-3 overflow-y-auto p-2">
          {messages.map((m, i) => <div key={i} className={`max-w-[80%] rounded p-3 text-sm ${m.role === 'user' ? 'ml-auto bg-primary text-primary-foreground' : 'bg-muted'}`}><pre className="whitespace-pre-wrap font-sans">{m.content}</pre></div>)}
          {pending && <div className="text-sm text-muted-foreground">Thinking…</div>}
        </div>
        <form className="flex gap-2 border-t pt-3" onSubmit={(e) => { e.preventDefault(); send(); }}>
          <Input value={input} onChange={(e) => setInput(e.target.value)} placeholder="Ask about your creatives, e.g. which hooks have the lowest CPA this month?" />
          <Button type="submit" disabled={pending}>Send</Button>
        </form>
      </section>
    </div>
  );
}
```

```tsx
'use client';
import { useEffect, useState } from 'react';

import type { GeneratedAsset } from '@/lib/types';

type Opt = { id: string; name: string };

export function PublishDialog({ asset, accounts, open, onOpenChange }: { asset: GeneratedAsset; accounts: Opt[]; open: boolean; onOpenChange: (o: boolean) => void }) {
  const [account, setAccount] = useState(''); const [campaigns, setCampaigns] = useState<Opt[]>([]); const [campaign, setCampaign] = useState('');
  const [adsets, setAdsets] = useState<Opt[]>([]); const [adset, setAdset] = useState(''); const [pageId, setPageId] = useState(''); const [link, setLink] = useState('');
  const [headline, setHeadline] = useState(asset.headline); const [text, setText] = useState(asset.primary_text); const [cta, setCta] = useState(asset.cta_type);
  const [status, setStatus] = useState<'PAUSED' | 'ACTIVE'>('PAUSED'); const [msg, setMsg] = useState<string | null>(null);

  useEffect(() => { if (account) api.campaigns(account).then(setCampaigns); else setCampaigns([]); setCampaign(''); }, [account]);
```
